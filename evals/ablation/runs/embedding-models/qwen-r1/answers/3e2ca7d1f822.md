### Analysis

The spec requires a single pure function `validate_inn` that checks: (1) the string is all digits, (2) length is 10 or 12, (3) the FNS control-digit checksum(s) pass. I'll implement the two checksum algorithms (one control digit for 10-digit, two for 12-digit) and write a focused pytest suite covering valid numbers, invalid checksums, and all the rejection cases from the spec.

### Implementation

```python
# validate_inn.py
"""Validate Russian INN (Идентификационный номер налогоплательщика).

10-digit INN — legal entity (юрлицо), one control digit.
12-digit INN — individual (физлицо), two control digits.
Algorithm per FNS (ФНС) order.
"""

# Weights for the 10-digit (legal entity) control digit.
_INN10_WEIGHTS: tuple[int, ...] = (2, 4, 10, 3, 5, 9, 4, 6, 8)

# Weights for the 12-digit (individual) first control digit.
_INN12_WEIGHTS_1: tuple[int, ...] = (7, 2, 4, 10, 3, 5, 9, 4, 6, 8)

# Weights for the 12-digit (individual) second control digit.
_INN12_WEIGHTS_2: tuple[int, ...] = (3, 7, 2, 4, 10, 3, 5, 9, 4, 6, 8)


def _checksum(digits: list[int], weights: tuple[int, ...]) -> int:
    """Compute the FNS control digit: (weighted_sum % 11) % 10."""
    return sum(d * w for d, w in zip(digits, weights)) % 11 % 10


def validate_inn(inn: str) -> bool:
    """Return True if *inn* is a valid Russian INN (10 or 12 digits).

    Rejects: empty strings, whitespace, non-digit characters,
    and any length other than 10 or 12.
    """
    # Fast rejection: must be non-empty, all digits, correct length.
    if not inn or not inn.isdigit():
        return False

    if len(inn) == 10:
        digits = [int(c) for c in inn]
        return digits[9] == _checksum(digits[:9], _INN10_WEIGHTS)

    if len(inn) == 12:
        digits = [int(c) for c in inn]
        return (
            digits[10] == _checksum(digits[:10], _INN12_WEIGHTS_1)
            and digits[11] == _checksum(digits[:11], _INN12_WEIGHTS_2)
        )

    return False
```

# test_validate_inn.py
import pytest

from validate_inn import validate_inn


# ── Valid INNs (verified against the FNS algorithm) ──────────────────────

class TestValid:
    def test_valid_10_digit(self):
        # Sberbank
        assert validate_inn("7707083893") is True

    def test_valid_12_digit(self):
        assert validate_inn("500100732259") is True

    def test_valid_10_digit_another(self):
        # 7710058590 — another known-valid legal-entity INN
        assert validate_inn("7710058590") is True


# ── Invalid checksums ─────────────────────────────────────────────────────

class TestInvalidChecksum:
    def test_10_digit_wrong_control(self):
        # Last digit changed from 3 → 4
        assert validate_inn("7707083894") is False

    def test_12_digit_wrong_first_control(self):
        # 11th digit changed from 5 → 6
        assert validate_inn("500100732659") is False

    def test_12_digit_wrong_second_control(self):
        # 12th digit changed from 9 → 0
        assert validate_inn("500100732250") is False


# ── Rejection: non-digit / whitespace / empty ────────────────────────────

class TestNonDigitRejection:
    @pytest.mark.parametrize("bad", [
        "",            # empty
        "   ",         # only spaces
        "770708389 ",  # trailing space
        " 7707083893", # leading space
        "770708389a",  # letter
        "770708389-",  # hyphen
        "770708389.3", # dot
        "770708389\n", # newline
        "770708389\t", # tab
    ])
    def test_rejects_non_digit_or_whitespace(self, bad: str):
        assert validate_inn(bad) is False


# ── Rejection: wrong length ───────────────────────────────────────────────

class TestWrongLength:
    @pytest.mark.parametrize("bad", [
        "123456789",      # 9 digits
        "12345678901",    # 11 digits
        "1234567890123",  # 13 digits
        "12345678901234", # 14 digits
        "1",              # 1 digit
    ])
    def test_rejects_wrong_length(self, bad: str):
        assert validate_inn(bad) is False

**Edge cases considered:**
- `str.isdigit()` returns `True` for Unicode digits (e.g. Arabic-Indic `٣`). If you need to reject those, replace the guard with `inn.isascii() and inn.isdigit()`. The spec says "любые нецифровые символы" so the current behaviour is correct for the stated requirement.
- The `% 11 % 10` double-mod is intentional per the FNS algorithm: if the intermediate sum mod 11 equals 10, the control digit is 0 (not 10).

**Complexity:** O(n) where n ∈ {10, 12} — effectively O(1). No allocations beyond the small digit list.
