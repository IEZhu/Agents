"""validate_inn — проверка российского ИНН (10/12 цифр) по алгоритму ФНС."""

_INN_10_W1 = (2, 4, 10, 3, 5, 9, 4, 6)
_INN_10_W2 = (7, 2, 4, 10, 3, 5, 9, 4, 6)
_INN_12_W1 = (2, 4, 10, 3, 5, 9, 4, 6, 8)
_INN_12_W2 = (7, 2, 4, 10, 3, 5, 9, 4, 6, 8)


def validate_inn(inn: str) -> bool:
    """Возвращает True, если *inn* — корректный российский ИНН.

    * 10 цифр — юридическое лицо
    * 12 цифр — физическое лицо (12-я цифра — сериальный номер, не проверяется)
    """
    if not inn or not inn.isdigit():
        return False

    if len(inn) == 10:
        return _check_10(inn)
    if len(inn) == 12:
        return _check_12(inn)
    return False


def _check_10(inn: str) -> bool:
    d = [int(c) for c in inn]
    k1 = sum(v * w for v, w in zip(d[:8], _INN_10_W1)) % 11 % 10
    k2 = sum(v * w for v, w in zip(d[:9], _INN_10_W2)) % 11 % 10
    return k1 == d[8] and k2 == d[9]


def _check_12(inn: str) -> bool:
    d = [int(c) for c in inn]
    k1 = sum(v * w for v, w in zip(d[:9], _INN_12_W1)) % 11 % 10
    k2 = sum(v * w for v, w in zip(d[:10], _INN_12_W2)) % 11 % 10
    return k1 == d[9] and k2 == d[10]
```

# validate_inn.py
"""Validation of Russian INN (ИНН) numbers per the FNS checksum algorithm."""

_INN_10_WEIGHTS = (2, 4, 10, 3, 5, 9, 4, 6, 8)
_INN_12_WEIGHTS_FIRST = (7, 2, 4, 10, 3, 5, 9, 4, 6, 8)
_INN_12_WEIGHTS_SECOND = (3, 7, 2, 4, 10, 3, 5, 9, 4, 6, 8)


def validate_inn(inn: str) -> bool:
    """Return True if *inn* is a well-formed Russian INN.

    - 10 digits → legal entity (юрлицо)
    - 12 digits → individual (физлицо)
    Any other length, non-digit characters, or failed checksum → False.
    """
    if not inn.isascii() or not inn.isdigit():
        return False

    if len(inn) not in (10, 12):
        return False

    digits = [int(c) for c in inn]

    if len(digits) == 10:
        return _checksum(digits, _INN_10_WEIGHTS) == digits[9]

    first  = _checksum(digits, _INN_12_WEIGHTS_FIRST)
    second = _checksum(digits, _INN_12_WEIGHTS_SECOND)
    return first == digits[10] and second == digits[11]


def _checksum(digits: list[int], weights: tuple[int, ...]) -> int:
    """FNS control digit: weighted sum mod 11, then mod 10."""
    return sum(d * w for d, w in zip(digits, weights)) % 11 % 10
```
