### Analysis

The spec requires a pure function `validate_inn(inn: str) -> bool` that validates Russian INN numbers (10-digit for legal entities, 12-digit for individuals) using the FNS control-digit algorithm. Rejection rules: non-digit characters, wrong length, empty string. I'll implement the two checksums and write pytest tests covering valid/invalid/injection cases.

### Implementation

```python
# validate_inn.py
"""Validation of Russian INN (ИНН) numbers per the FNS checksum algorithm."""

_INN_10_WEIGHTS = (2, 4, 10, 3, 5, 9, 4, 6, 8)
_INN_12_WEIGHTS_FIRST = (7, 2, 4, 10, 3, 5, 9, 4, 6, 8)
_INN_12_WEIGHTS_SECOND = (3, 7, 2, 4, 10, 3, 5, 9, 4, 6, 8)


def validate_inn(inn: str) -> bool:
    """Return True if *inn* is a well-formed Russian INN.

    - 10 digits → legal entity (юрлицо)
    - 12 digits → individual (физлицо)
    Any other length, non-digit characters, or failed checksum → False.
    """
    if not inn.isascii() or not inn.isdigit():
        return False

    if len(inn) not in (10, 12):
        return False

    digits = [int(c) for c in inn]

    if len(digits) == 10:
        return _checksum(digits, _INN_10_WEIGHTS) == digits[9]

    first  = _checksum(digits, _INN_12_WEIGHTS_FIRST)
    second = _checksum(digits, _INN_12_WEIGHTS_SECOND)
    return first == digits[10] and second == digits[11]


def _checksum(digits: list[int], weights: tuple[int, ...]) -> int:
    """FNS control digit: weighted sum mod 11, then mod 10."""
    return sum(d * w for d, w in zip(digits, weights)) % 11 % 10
```

import pytest
from validate_inn import validate_inn


class TestValidINN:
    def test_10_digit_legal_entity(self):
        assert validate_inn("7707083893") is True

    def test_12_digit_individual(self):
        assert validate_inn("500100000199") is True


class TestInvalidChecksum:
    def test_10_digit_wrong_control(self):
        assert validate_inn("7707083894") is False  # last digit off by 1

    def test_12_digit_wrong_first_control(self):
        assert validate_inn("500100000099") is False  # 11th digit wrong

    def test_12_digit_wrong_second_control(self):
        assert validate_inn("500100000198") is False  # 12th digit wrong


class TestFormatRejection:
    @pytest.mark.parametrize("inn", [
        "",            # empty
        " ",           # single space
        "77070838 93", # embedded space
        " 7707083893", # leading space
        "7707083893 ", # trailing space
        "770708389a",  # letter
        "770708389-",  # hyphen
        "770708389.3", # dot
        "7707083893\n",# newline
        "٧707083893",  # Arabic-Indic digit (non-ASCII)
    ])
    def test_rejects_bad_characters(self, inn):
        assert validate_inn(inn) is False

    @pytest.mark.parametrize("inn", [
        "123",
        "12345678",
        "123456789",
        "12345678901",
        "1234567890123",
    ])
    def test_rejects_wrong_length(self, inn):
        assert validate_inn(inn) is False
